{\large Four assumptions carry this design}
\end{center}
\vfill
\assumptionrow{1}{Language detail}{Language captures enough nuance to separate the target from clutter.}
\vspace{0.4em}
\assumptionrow{2}{Consistent interpretation}{Foundation models map the same phrase to the same concept across their backbones.}
\vspace{0.4em}
\assumptionrow{3}{Reliable anchors}{Cross-view segmentation can happen on a handful of moments in the video.}
\vspace{0.4em}
\assumptionrow{4}{Offline prediction}{Cross-view segmentation can happen forward and backward in time.}

\begin{frame}[t]
\begin{center}
{\large Stage 1: Source Frame Selection}
\end{center}
\vfill
  \begin{center}
  \includegraphics[width=0.8\textwidth]{imgs/ch3/block-1-focus.png}
  \end{center}
  Visibility Scorer computes score $s_\tau$:
  \[
    s_\tau = 0.99\,a_\tau + 0.01\,c_\tau, \qquad
    a_\tau = \frac{A_\tau}{H \cdot W}, \qquad
    c_\tau =
    1 -
    \frac{\sqrt{\left(c_{y,\tau} - \frac{H}{2}\right)^2 +
    \left(c_{x,\tau} - \frac{W}{2}\right)^2}
    }{\sqrt{\left(\frac{H}{2}\right)^2 +
    \left(\frac{W}{2}\right)^2}},
  \]
  \hspace{4mm} where $a_\tau$ is the normalised mask area and $c_\tau$ is the mask's centrality.
\vfill
\end{frame}

\begin{frame}[t]

{\large The four assumptions, revisited}
\end{center}
\vfill
\verdictrow{1}{Language detail}{\opentag{TBD}}{The bottleneck is perception, not description: the VLM cannot resolve a target that fills a handful of pixels. Foveated tokenisation is a promising remedy.}
\vspace{0.25em}
\verdictrow{2}{Consistent interpretation}{\heldtag}{Independently trained backbones map the same phrase to the same concept; once the anchor is correct, quality reaches the state-of-the-art band.}
\vspace{0.25em}
\verdictrow{3}{Reliable anchors}{\opentag{Re-evaluate}}{A handful of anchors completes the track, yet a single wrong anchor propagates faithfully across the entire take.}
\vspace{0.25em}
\verdictrow{4}{Offline prediction}{\heldtag$^{*}$}{Bidirectional propagation raises no failures.}
\vspace{0.4em}
\begin{center}
{\footnotesize $^{*}$Live applications require online prediction. Once the bridge is built only Stage 4 runs, so the pipeline still extends to live settings.}
